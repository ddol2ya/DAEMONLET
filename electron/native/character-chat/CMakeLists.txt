cmake_minimum_required(VERSION 3.18)
project(daemonlet_runtime_host LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This owner is only for Windows x64")
endif()
add_executable(daemonlet-runtime-host host.cpp)
target_compile_features(daemonlet-runtime-host PRIVATE cxx_std_17)
set_property(TARGET daemonlet-runtime-host PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_options(daemonlet-runtime-host PRIVATE /W4 /utf-8)
